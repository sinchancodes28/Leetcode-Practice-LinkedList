/* Structure of Linked List Node
class Node {
    int data;
    Node next;

    Node(int x) {
        data = x;
        next = null;
    }
}*/

class Solution {
    public int lengthOfLoop(Node head) {
        // code here
        Node slow = head;
        Node fast = head;
        while( fast != null && fast.next != null){
            slow = slow.next;
            fast = fast.next.next;
            if( slow == fast){
                // meeting node se length nikl li 
                return count(slow);
            }
        }
        return 0;
    }
    private int count(Node meetingNode){
        int count = 1;
        Node temp = meetingNode.next;
        while(temp != meetingNode){
            count++;
            temp = temp.next;
        }
        return count;
    }
}
